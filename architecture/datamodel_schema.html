Table / View Type Primary Key Key Columns Notes
entity Base Table imo_number CHAR(7) created_at, first_seen_at, first_source_id Anchor table — one row per IMO number. All other vessel data links here.
attr_value Partitioned (RANGE sys_from) av_id BIGSERIAL imo_number, attr_id, value_text/numeric/boolean/date/json, source_id, confidence, valid_from/to, sys_from/to, workflow_state, run_id, qc_flags INSERT-ONLY (PostgreSQL RULE). v_current Materialized View Unique on imo_number Pivots EAV rows (CASE/MAX) into ~50 wide columns matching vessels.js VESSELS property names. Joins latest_position for current lat/lon/speed. Refreshed concurrently via registry.refresh_all_current_views().
-- Insert-only rule on vessel.attr_value CREATE RULE attr_value_insert_only AS ON UPDATE TO vessel.attr_value DO INSTEAD INSERT INTO vessel.attr_value (imo_number, attr_id, value_text, source_id, valid_from, valid_to, sys_from, sys_to, workflow_state) VALUES (NEW.imo_number, NEW.attr_id, NEW.value_text, NEW.source_id, NEW.valid_from, NEW.valid_to, now(), 'infinity', NEW.workflow_state);

-- Spatial query example: vessels near a chokepoint (within 50 nautical miles) SELECT v.imo_number, v.vessel_name, lp.speed_kn, lp.nav_status FROM vessel.latest_position lp JOIN vessel.v_current v USING (imo_number) JOIN geo.choke_point cp ON cp.name = 'Strait of Hormuz' WHERE ST_DWithin(lp.geom::geography, cp.geom::geography, 92600); -- 50 nmi in metres
